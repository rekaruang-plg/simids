function targetKeys(){return ['pusdatinBirthMale','pusdatinBirthFemale','pusdatinSurvivingMale','pusdatinSurvivingFemale','localBirthMale','localBirthFemale','localSurvivingMale','localSurvivingFemale']}

function eventIdsFor(vaccineId,village,month,sex){
  const year=Number(state.settings.year),allowedChildren=state.children.filter(c=>(village==='all'||c.village===village)&&(!sex||c.sex===sex)),allowedIds=new Set(allowedChildren.map(c=>c.id));
  const eligible=(e)=>e.validated!==false&&allowedIds.has(e.childId)&&new Date(e.date+'T00:00:00').getFullYear()===year&&new Date(e.date+'T00:00:00').getMonth()+1<=Number(month);
  if(vaccineId==='HB0_TOTAL')return new Set(state.events.filter(e=>eligible(e)&&['HB0','HB0_24','HB0_1_7'].includes(e.vaccineId)).map(e=>e.childId));
  if(vaccineId==='HB0_24'||vaccineId==='HB0_1_7'){
    const want24=vaccineId==='HB0_24';
    return new Set(state.events.filter(e=>{
      if(!eligible(e))return false;
      if(e.vaccineId===vaccineId)return true;
      if(e.vaccineId!=='HB0')return false;
      const c=state.children.find(x=>x.id===e.childId);if(!c?.dob)return false;
      const days=Math.floor((new Date(e.date+'T00:00:00')-new Date(c.dob+'T00:00:00'))/86400000);
      return want24?days===0:(days>=1&&days<=7);
    }).map(e=>e.childId));
  }
  return new Set(state.events.filter(e=>eligible(e)&&e.vaccineId===vaccineId).map(e=>e.childId))
}
